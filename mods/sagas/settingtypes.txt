# How rare landmarks are. 1 = as designed by each landmark, 0.1 = ten times more often, 10 = ten times rarer.
sagas_rarity_scale (Rarity scale) float 1.0 0.001 1000

# Comma-separated sagas (e.g. "antiquity") or landmark ids (e.g. "sagas_ruins:cathedral") that are never placed automatically.
sagas_disabled (Disabled sagas or landmarks) string

# At most this many landmarks per generated mapchunk.
sagas_max_per_chunk (Sagas per mapchunk) int 1 1 4

# Place landmarks that are larger than a mapchunk (they are emerged and placed after generation).
sagas_large_enable (Large landmarks) bool true

# Show a landmark's story when a player enters it for the first time.
sagas_lore (Show lore) bool true

# Use the game's flowing water inside landmarks instead of still water.
sagas_real_water (Flowing water in landmarks) bool false

# Never place landmarks during map generation (/saga place still works).
sagas_disable_mapgen (No automatic placement) bool false

# Log placements and mark candidate sites.
sagas_debug (Debug output) bool false

# Development: also test every landmark's site in every generated mapchunk without placing it, and
# count fits and rejection reasons (/saga stats, avanti survey). Slows map generation down.
sagas_survey (Survey site fit) bool false

# Development: place nothing but these landmarks (comma-separated ids, e.g. "sagas_ruins:mammoth_skeleton"),
# to see in a fresh world where map generation puts them. Combine with a low sagas_rarity_scale.
sagas_only (Only these landmarks) string

# Development: a /saga command run for the singleplayer after joining, e.g. "find mammoth_skeleton" or
# "tour" (Saga Forge sets it to land on a landmark).
sagas_start (Start with) string
